<script>
window.addEventListener('load', function() {
  // Helper: Escape HTML characters for the preview text
  function escapeHtml(text) {
    var map = {
      '&': '&amp;',
      '<': '&lt;',
      '>': '&gt;',
      '"': '&quot;',
      "'": '&#039;'
    };
    return text.replace(/[&<>"']/g, function(m) { return map[m]; });
  }

  try {
    // 1. Symbol Collapser (Specific Pattern: $Var<-[...])
    document.querySelectorAll('td').forEach(function(td) {
      var html = td.innerHTML.trim();
      var match = html.match(/^(\$\w+)\s*(&lt;-|<-|-&gt;|->)\s*\[([\s\S]*)\]$/);
      if (match) {
        var sym = match[1];
        var arrow = match[2].replace('&lt;', '<').replace('&gt;', '>');
        var content = match[3];
        
        if (content.length > 50) {
           td.innerHTML = '<details><summary style="cursor:pointer;color:#007bff;">' + 
                          sym + ' ' + arrow + ' (Show Value)</summary>' +
                          '<div style="max-height:300px;overflow:auto;background:#f8f9fa;border:1px solid #ddd;padding:5px;white-space:pre-wrap;font-family:monospace;">' + 
                          content + '</div></details>';
        }
      }
    });

    // 2. Long Content Collapser for Slim Tables (Generic Pattern: > 500 chars in slimRowColorX)
    var slimRows = document.querySelectorAll('tr[class^="slimRowColor"]');
    slimRows.forEach(function(row) {
      row.querySelectorAll('td').forEach(function(td) {
        // Skip if already processed (e.g. by Symbol Collapser)
        if (td.querySelector('details') || td.querySelector('.long-content-wrapper')) return;

        var txt = td.textContent;
        if (txt.length > 500) {
           var fullHtml = td.innerHTML;
           // Check for line breaks (newlines in text content or <br> tags in HTML)
           // We use textContent for \n to catch actual line breaks in the text, and innerHTML for <br> tags
           var hasLineBreaks = txt.indexOf('\n') !== -1 || fullHtml.indexOf('<br') !== -1;
           var style = hasLineBreaks ? 'white-space: pre-wrap; font-family: monospace; font-size: medium;' : '';
           
           // Create a safe preview (escaped)
           var previewText = escapeHtml(txt.substring(0, 500));
           
           // We use a wrapper to toggle between preview and full content
           td.innerHTML = '<div class="long-content-wrapper">' + 
                          '<div class="preview" style="' + style + '">' + 
                          previewText + '... <a href="javascript:void(0)" class="toggle-link expand" style="color:#007bff;">[Show More]</a></div>' + 
                          '<div class="full-content" style="display:none; ' + style + '">' + 
                          fullHtml + 
                          ' <a href="javascript:void(0)" class="toggle-link collapse" style="color:#007bff;">[Show Less]</a></div>' + 
                          '</div>';
           
           // Bind click event for both links
           var wrapper = td.querySelector('.long-content-wrapper');
           wrapper.querySelectorAll('.toggle-link').forEach(function(link) {
               link.onclick = function() {
                   var isExpand = this.classList.contains('expand');
                   wrapper.querySelector('.preview').style.display = isExpand ? 'none' : 'block';
                   wrapper.querySelector('.full-content').style.display = isExpand ? 'block' : 'none';
               };
           });
        }
      });
    });

  } catch (e) {
    console.error("FitNesse Custom Script Error:", e);
  }
});
</script>
